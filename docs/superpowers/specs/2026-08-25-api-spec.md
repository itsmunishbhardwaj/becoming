## 1. Design decisions

### Direct client SDK, no BFF layer

Supabase client SDK + RLS handles auth isolation automatically. Every query from
the browser is scoped to `auth.uid()` via RLS policies — no API routes needed
for CRUD. The store wraps Supabase client calls directly.

API routes are used only for:
- Auth callback (`/auth/callback`) — required by OAuth flow
- LLM proxy (`/api/llm`) — keeps API key off the client

### Store interface is unchanged

`src/data/store.js` keeps the same exported function signatures. Only the
implementation changes — `vaultStore` is replaced by `supabaseStore`. Every
screen that calls `listGoals()`, `appendLog()`, etc. is unaware of the change.

New functions are added for habit goals (no equivalent in vault backend):
- `toggleHabitDay(goalId, date)` 
- `getHabitLogs(goalId, year)`

### Session via cookies (SSR-compatible)

Supabase `@supabase/ssr` package stores the session in cookies, not
`localStorage`. This means server components and middleware can read auth state
without a client-side round trip.

## 3. Supabase clients

### Browser client — `src/lib/supabase/client.js`

```js
import { createBrowserClient } from '@supabase/ssr'

export function createClient() {
  return createBrowserClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY
  )
}
```

Called once per component that needs it, or via a shared context. The browser
client reads the session from cookies automatically.

### Server client — `src/lib/supabase/server.js`

```js
import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'

export function createClient() {
  const cookieStore = cookies()
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY,
    {
      cookies: {
        getAll() { return cookieStore.getAll() },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) =>
            cookieStore.set(name, value, options)
          )
        },
      },
    }
  )
}
```

Used in server components, API route handlers, and middleware.

## 4. Auth middleware — `middleware.js`

Protects all routes except `/login` and `/auth/callback`. Refreshes the session
cookie on every request (required by Supabase SSR to keep sessions alive).

```js
import { createServerClient } from '@supabase/ssr'
import { NextResponse } from 'next/server'

export async function middleware(request) {
  let response = NextResponse.next({ request })
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY,
    {
      cookies: {
        getAll() { return request.cookies.getAll() },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            request.cookies.set(name, value)
            response = NextResponse.next({ request })
            response.cookies.set(name, value, options)
          })
        },
      },
    }
  )

  const { data: { user } } = await supabase.auth.getUser()

  const { pathname } = request.nextUrl
  const isPublic = pathname.startsWith('/login') || pathname.startsWith('/auth')

  if (!user && !isPublic) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  return response
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico|.*\\.png$).*)'],
}
```

## 5. Auth flow — login page & callback

### Login page — `app/(auth)/login/page.jsx`

```jsx
'use client'
import { createClient } from '../../lib/supabase/client'

export default function LoginPage() {
  async function signInWithGoogle() {
    const supabase = createClient()
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: `${location.origin}/auth/callback` },
    })
  }

  return (
    <main>
      <h1>Becoming</h1>
      <button onClick={signInWithGoogle}>Continue with Google</button>
    </main>
  )
}
```

Styled to match the paper theme — exact implementation is UI spec territory.

### OAuth callback — `app/auth/callback/route.js`

```js
import { createClient } from '../../../lib/supabase/server'
import { NextResponse } from 'next/server'

export async function GET(request) {
  const { searchParams, origin } = new URL(request.url)
  const code = searchParams.get('code')
  if (code) {
    const supabase = createClient()
    await supabase.auth.exchangeCodeForSession(code)
  }
  return NextResponse.redirect(`${origin}/`)
}
```

## 7. Data shape mapping

Supabase returns snake_case columns. JS objects use camelCase. The mapping
layer lives in `supabaseStore.js`.

### Goal: Supabase row → JS object

```
goals row + rounds rows
  ↓ rowToGoal()
{
  id:             string (uuid)
  name:           string
  cat:            string
  color:          string | undefined
  type:           "tracker" | "habit"
  state:          string
  dormantNote:    string | undefined   ← dormant_note
  retro:          string | undefined
  baseline:       jsonb value
  target:         jsonb value
  endDate:        string               ← end_date (ISO)
  currentRound:   number              ← current_round
  createdAt:      string               ← created_at
  ambition:       string
  howWeGetThere:  string              ← how_we_get_there
  indicators: {
    right:  string[]                  ← indicators_right
    wrong:  string[]                  ← indicators_wrong
    stall:  string[]                  ← indicators_stall
  }
  rounds: [{ n, targetValue, startDate, endDate }]
          ← rounds rows: n, target_value, start_date, end_date
}
```

### Log: Supabase rows → JS object

```
log_events rows + log_notes rows for a date
  ↓ rowsToLog()
{
  date:   string
  events: [{ verb, goalId, time?, durationMin?, payload? }]
          ← log_events: verb, goal_id→goalId, time, duration_min→durationMin, payload
  notes:  { [goalId]: text }
          ← log_notes: goal_id→goalId, text
}
```

## 8. supabaseStore implementation

### `listGoals()`

```js
const { data } = await supabase
  .from('goals')
  .select('*, rounds(*)')
  .order('created_at', { ascending: true })
return (data || []).map(rowToGoal)
```

### `getGoal(id)`

```js
const { data } = await supabase
  .from('goals')
  .select('*, rounds(*)')
  .eq('id', id)
  .single()
return data ? rowToGoal(data) : null
```

### `saveGoal(goal)`

Two-step: upsert the goal row, then replace its rounds.

```js
// 1. Upsert goal
await supabase.from('goals').upsert(goalToRow(goal))

// 2. Replace rounds: delete old, insert new
await supabase.from('rounds').delete().eq('goal_id', goal.id)
if (goal.rounds?.length) {
  await supabase.from('rounds').insert(
    goal.rounds.map(r => ({
      goal_id:      goal.id,
      n:            r.n,
      target_value: r.targetValue,
      start_date:   r.startDate,
      end_date:     r.endDate,
    }))
  )
}
```

No transaction primitive in the Supabase JS client. Delete + insert is safe at
personal scale — concurrent writes from two devices on the same goal are
unlikely and last-write-wins is acceptable (matching current vault behavior).

### `readLog(date)`

```js
const [eventsRes, notesRes] = await Promise.all([
  supabase.from('log_events').select('*').eq('date', date),
  supabase.from('log_notes').select('*').eq('date', date),
])
const events = eventsRes.data || []
const notes = notesRes.data || []
if (!events.length && !notes.length) return null
return rowsToLog(date, events, notes)
```

### `readLogsInRange({ from, to })`

```js
const [eventsRes, notesRes] = await Promise.all([
  supabase.from('log_events').select('*').gte('date', from).lte('date', to),
  supabase.from('log_notes').select('*').gte('date', from).lte('date', to),
])
// Group by date, assemble Log objects
const byDate = {}
for (const e of eventsRes.data || []) {
  if (!byDate[e.date]) byDate[e.date] = { events: [], notes: [] }
  byDate[e.date].events.push(e)
}
for (const n of notesRes.data || []) {
  if (!byDate[n.date]) byDate[n.date] = { events: [], notes: [] }
  byDate[n.date].notes.push(n)
}
return Object.entries(byDate).map(([date, { events, notes }]) =>
  rowsToLog(date, events, notes)
)
```

### `appendLog(date, event)`

Idempotency: check for existing matching row before inserting.

```js
const { data: existing } = await supabase
  .from('log_events')
  .select('id')
  .eq('goal_id', event.goalId)
  .eq('date', date)
  .eq('verb', event.verb)
  .eq('time', event.time ?? null)
  .eq('payload', event.payload ?? null)
  .maybeSingle()

if (!existing) {
  await supabase.from('log_events').insert({
    goal_id:      event.goalId,
    date,
    verb:         event.verb,
    time:         event.time ?? null,
    duration_min: event.durationMin ?? null,
    payload:      event.payload ?? null,
  })
}
```

### `deleteLogEvent(date, event)`

```js
let q = supabase
  .from('log_events')
  .delete()
  .eq('goal_id', event.goalId)
  .eq('date', date)
  .eq('verb', event.verb)

if (event.time != null) q = q.eq('time', event.time)
else q = q.is('time', null)

await q
```

### `saveNote(date, goalId, text)`

```js
if (!text.trim()) {
  await supabase.from('log_notes').delete()
    .eq('goal_id', goalId).eq('date', date)
} else {
  await supabase.from('log_notes').upsert({
    goal_id: goalId, date, text: text.trim(),
    updated_at: new Date().toISOString(),
  })
}
```

### `toggleHabitDay(goalId, date)` (new)

```js
const { data: existing } = await supabase
  .from('habit_logs')
  .select('id, kept')
  .eq('goal_id', goalId)
  .eq('date', date)
  .maybeSingle()

if (existing?.kept) {
  // Already kept — unmark (set kept=false, not delete, so the row persists)
  await supabase.from('habit_logs').update({ kept: false })
    .eq('id', existing.id)
} else if (existing) {
  // Row exists but kept=false — re-mark
  await supabase.from('habit_logs').update({ kept: true })
    .eq('id', existing.id)
} else {
  // No row — insert kept=true
  await supabase.from('habit_logs').insert({ goal_id: goalId, date, kept: true })
}
```

### `getHabitLogs(goalId, year)` (new)

Returns array of ISO date strings where `kept = true`.

```js
const from = `${year}-01-01`
const to   = `${year}-12-31`
const { data } = await supabase
  .from('habit_logs')
  .select('date')
  .eq('goal_id', goalId)
  .eq('kept', true)
  .gte('date', from)
  .lte('date', to)
return (data || []).map(r => r.date)
```

## 9. LLM proxy — `app/api/llm/route.js`

Moves the existing `llmMiddleware.js` logic to a Vercel serverless function.
API key stays server-side. Client calls `fetch('/api/llm', ...)` — same URL
the app uses today.

```js
export async function POST(request) {
  const body = await request.json()
  const res = await fetch(process.env.LLM_BASE_URL + '/chat/completions', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${process.env.LLM_API_KEY}`,
    },
    body: JSON.stringify(body),
  })
  const data = await res.json()
  return Response.json(data, { status: res.status })
}
```

`LLM_BASE_URL` and `LLM_API_KEY` are Vercel environment variables — never
exposed to the client.
